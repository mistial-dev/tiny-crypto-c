cmake_minimum_required(VERSION 3.24)
project(target_consumer LANGUAGES C)
set(TINY_CRYPTO_BUILD_TESTS OFF)
set(TINY_CRYPTO_BUILD_BENCHMARKS OFF)
add_subdirectory("${SOURCE_DIR}" library)
add_executable(target_check target.c "${SOURCE_DIR}/tests/support/munit.c"
  "${SOURCE_DIR}/tests/support/test_util.c" "${SOURCE_DIR}/tests/support/cavp.c")
set_target_properties(target_check PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_include_directories(target_check PRIVATE "${SOURCE_DIR}/tests/support")
target_link_libraries(target_check PRIVATE tiny-crypto-c::tiny-crypto-c)
foreach(target full piv twic desfire)
  string(TOUPPER "${target}" name)
  if(TINY_CRYPTO_TARGET STREQUAL target)
    target_compile_definitions(target_check PRIVATE EXPECT_${name}=1)
  else()
    target_compile_definitions(target_check PRIVATE EXPECT_${name}=0)
  endif()
endforeach()
enable_testing()
add_test(NAME target_runtime COMMAND target_check)
