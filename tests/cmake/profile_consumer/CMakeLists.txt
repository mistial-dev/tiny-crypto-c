cmake_minimum_required(VERSION 3.24)
project(profile_consumer C)
set(TINY_CRYPTO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TINY_CRYPTO_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCE_DIR}" product)
foreach(kind configured direct)
  add_library(profile_${kind} OBJECT check.c)
  target_compile_definitions(profile_${kind} PRIVATE
    EXPECT_PROFILE=${EXPECT_PROFILE} EXPECT_CAPABILITIES=${EXPECT_CAPABILITIES}
    EXPECT_WIDE=${EXPECT_WIDE} EXPECT_TINY=${EXPECT_TINY} EXPECT_GHASH=${EXPECT_GHASH})
  if(kind STREQUAL "configured")
    target_link_libraries(profile_${kind} PRIVATE tiny-crypto-c)
  else()
    target_include_directories(profile_${kind} PRIVATE "${SOURCE_DIR}/src")
    target_compile_definitions(profile_${kind} PRIVATE TC_RESOURCE_PROFILE=${EXPECT_PROFILE})
  endif()
  if(OVERRIDE_WIDE)
    target_compile_definitions(profile_${kind} PRIVATE EXPECT_OVERRIDE=1)
    if(kind STREQUAL "direct")
      target_compile_definitions(profile_${kind} PRIVATE TC_AES_WIDE_OPS=0)
    endif()
  endif()
endforeach()
